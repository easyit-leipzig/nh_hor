# PATH_INFO nach PHP-Dateien unterbinden
AcceptPathInfo Off

RewriteEngine On

# Falsche Doppeladresse dauerhaft auf die Root-Prüfdatei umleiten
RewriteCond %{THE_REQUEST} \s/+check_db\.php/checkdb\.php[\s?] [NC]
RewriteRule ^check_db\.php/checkdb\.php$ /checkdb.php [R=302,L]

# Alternative Schreibweise ebenfalls auf die neue Datei umleiten
RewriteRule ^check_db\.php$ /checkdb.php [R=302,L]
